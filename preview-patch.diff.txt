+/**
+ * Infer JSON Schema type from a runtime value.
+ * Used to build response schemas from discovery.exampleOutput.
+ */
+function inferSchema(value: unknown): Record<string, unknown> {
+  if (value === null) return { type: "null" };
+  if (typeof value === "boolean") return { type: "boolean" };
+  if (typeof value === "number") return { type: "number" };
+  if (typeof value === "string") return { type: "string" };
+  if (Array.isArray(value)) {
+    const items = value.length > 0 ? inferSchema(value[0]) : { type: "object" };
+    return { type: "array", items };
+  }
+  if (typeof value === "object") {
+    const properties: Record<string, unknown> = {};
+    for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
+      properties[k] = inferSchema(v);
+    }
+    return { type: "object", properties };
+  }
+  return { type: "string" };
+}

+/**
+ * MCP (Model Context Protocol) manifest for agent tool discovery.
+ * Exposes x402-paid tools with input schemas and payment metadata.
+ *
+ * @see https://modelcontextprotocol.io
+ */
+export async function GET() {
+  const base = serverEnv.PUBLIC_BASE_URL ?? "";
+
+  const tools = VENDING_SERVICES.filter((s) => s.enabled).map((s) => {
+    const properties: Record<string, Record<string, unknown>> = {};
+    const required: string[] = [];
+
+    for (const p of s.queryParams) {
+      properties[p.name] = {
+        type: "string",
+        description: p.description ?? p.name,
+      };
+      if (p.required) required.push(p.name);
+    }
+
+    return {
+      name: s.slug,
+      description: s.description,
+      category: s.category ?? "atom",
+      pricing: {
+        price_usd: s.price,
+        scheme: s.scheme,
+        network: serverEnv.X402_NETWORK_MODE,
+        asset: "USDC",
+      },
+      inputSchema: {
+        type: "object",
+        properties,
+        ...(required.length ? { required } : {}),
+      },
+      ...(s.discovery?.exampleQuery
+        ? { exampleInput: s.discovery.exampleQuery }
+        : {}),
+      ...(s.discovery?.exampleOutput
+        ? { exampleOutput: s.discovery.exampleOutput }
+        : {}),
+      ...(s.discovery?.estimatedLatencyMs
+        ? { estimatedLatencyMs: s.discovery.estimatedLatencyMs }
+        : {}),
+      ...(s.discovery?.tags ? { tags: s.discovery.tags } : {}),
+      endpoint: {
+        method: "GET",
+        url: `${base}/api/v/${s.slug}`,
+      },
+    };
+  });
+
+  return NextResponse.json({
+    schemaVersion: "v1",
+    name: "x402-vending-machine",
+    description:
+      "Base mainnet x402 bundler hub. Pay-per-call utilities, bundles, and Kronos research forecasts.",
+    network: serverEnv.X402_NETWORK_MODE,
+    pay_to: serverEnv.X402_PAY_TO_ADDRESS,
+    facilitator: serverEnv.X402_FACILITATOR_URL,
+    tools,
+  });
